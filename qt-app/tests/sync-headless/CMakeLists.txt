cmake_minimum_required(VERSION 3.16)
project(MegaVaultSyncHeadless LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt5 5.12 QUIET COMPONENTS Core Network)
if(Qt5_FOUND)
 set(SYNC_QT Qt5)
else()
 find_package(Qt6 REQUIRED COMPONENTS Core Network)
 set(SYNC_QT Qt6)
endif()
add_executable(sync_protocol_tests ../sync_protocol.cpp ../../src/sync_protocol.cpp)
target_include_directories(sync_protocol_tests PRIVATE ../../include)
target_link_libraries(sync_protocol_tests PRIVATE ${SYNC_QT}::Core ${SYNC_QT}::Network)
enable_testing()
add_test(NAME sync_protocol COMMAND sync_protocol_tests)

# Encrypted ledger tests do not require the core, GUI, FFmpeg, or SQLCipher.
find_path(SYNC_SODIUM_INCLUDE sodium.h)
find_library(SYNC_SODIUM_LIBRARY NAMES sodium libsodium)
if(SYNC_SODIUM_INCLUDE AND SYNC_SODIUM_LIBRARY)
 add_executable(sync_state_tests ../sync_state.cpp ../../src/sync_state.cpp ../../src/sync_protocol.cpp)
 target_include_directories(sync_state_tests PRIVATE ../../include ${SYNC_SODIUM_INCLUDE})
 target_link_libraries(sync_state_tests PRIVATE ${SYNC_QT}::Core ${SYNC_SODIUM_LIBRARY})
 add_test(NAME sync_state COMMAND sync_state_tests)
else()
 message(WARNING "libsodium not found: only protocol tests will be built; encrypted ledger tests are skipped")
endif()
